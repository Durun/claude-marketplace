結論: 共有リンクでメモを開けるのは、メモが published で、リンクの期限内のときだけにする。

// apps/api/src/share/open.ts
app.get("/s/:token", async (req, res) => {
  const link = await shareLinkRepository.findByToken(req.params.token);
  if (!link) return res.sendStatus(404);
  if (link.expiresAt <= new Date()) return res.sendStatus(410);
  const note = await noteRepository.findById(link.noteId);
  if (note?.status !== "published") return res.sendStatus(404);
  return res.json(note);
});
